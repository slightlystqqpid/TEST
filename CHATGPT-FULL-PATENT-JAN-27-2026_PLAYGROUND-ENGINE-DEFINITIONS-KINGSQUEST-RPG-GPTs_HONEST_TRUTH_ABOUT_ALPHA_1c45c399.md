import os, hashlib, csv

SCAN_ROOTS = [
    r"D:\SYZYGY",
    r"D:\archives",
    r"D:\redeemer",
    r"D:\exports",
    r"D:\experiments"
]

OUT = "syzygy_file_fingerprint.csv"

def hash_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024*1024), b""):
            h.update(chunk)
    return h.hexdigest()

rows = []

for root in SCAN_ROOTS:
    for dirpath, _, files in os.walk(root):
        for name in files:
            path = os.path.join(dirpath, name)
            try:
                size = os.path.getsize(path)
                h = hash_file(path)
                rows.append([h, size, path])
            except:
                pass

with open(OUT, "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["hash", "size", "path"])
    writer.writerows(rows)

print("Scan complete:", OUT)
