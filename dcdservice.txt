    Private Function Crypt(ByRef pass As String, ByRef Strg As String) As String

        Dim b As Integer
        Dim j As String
        Dim h As String
        Dim i As Integer
        Dim a As Integer
        Dim tmp1 As String
        Dim tmp2 As String

        a = 1

        For i = 1 To Len(Strg)
            b = Asc(Mid(pass, a, 1))
            a = a + 1
            If a > Len(pass) Then a = 1
            tmp1 = Chr(Asc(Mid(Strg, i, 1)) Xor b)
            tmp2 = tmp2 & tmp1
        Next

        Strg = tmp2
        h = ""

        For i = 1 To Len(Strg)
            j = Hex(Asc(Mid(Strg, i, 1)))
            If Len(j) = 1 Then j = "0" & j
            h = h & j
        Next

        Return h

    End Function

    <WebMethod()> Public Function AddDocument(ByVal projid As Integer, ByVal filename As String, ByVal category As String, ByVal pplid As Integer) As String

        Dim SqlConn As New SqlConnection(strVcnetConn)
        Dim cnt As Integer = 0
        Dim docid As Integer = 0
        Dim sqlstmt As String
        Dim strcat As String

        Try
            SqlConn.Open()

            'check if document has already been added
            sqlstmt = "SELECT count(*) FROM documents " & _
             "WHERE projectid = " & projid & " " & _
              "AND docfilename = '" & filename & "'"

            Dim sqlCmd1 As New SqlCommand(sqlstmt, SqlConn)
            sqlCmd1.CommandType = CommandType.Text
            cnt = CInt(sqlCmd1.ExecuteScalar.ToString)

            If cnt = 0 Then 'add document if not already in database
                sqlstmt = LCase("INSERT INTO documents (projectid, docfilename, documentcategory, peopleid)  " & _
                    "VALUES (" & projid & ", '" & filename & "', '" & category & "', " & pplid & ")")

                Dim sqlCmd2 As New SqlCommand(sqlstmt, SqlConn)
                sqlCmd2.CommandType = CommandType.Text
                sqlCmd2.ExecuteNonQuery()
                strcat = category
            Else 'check if document category
                sqlstmt = "SELECT documentcategory FROM documents " & _
                 "WHERE projectid = " & projid & " " & _
                  "AND docfilename = '" & filename & "'"

                Dim sqlCmd2 As New SqlCommand(sqlstmt, SqlConn)
                sqlCmd2.CommandType = CommandType.Text
                strcat = sqlCmd2.ExecuteScalar.ToString
            End If

            If strcat = category Then 'get documentid 
                sqlstmt = "SELECT documentid FROM documents " & _
                 "WHERE projectid = " & projid & " " & _
                  "AND docfilename = '" & filename & "'"

                Dim sqlCmd3 As New SqlCommand(sqlstmt, SqlConn)
                sqlCmd3.CommandType = CommandType.Text
                docid = CInt(sqlCmd3.ExecuteScalar.ToString)

                Return docid.ToString
            Else
                Return "Dup"
            End If

        Catch ex As SqlException
            Return ex.Number & "^" & ex.Message

        Finally
            If SqlConn.State = ConnectionState.Open Then
                SqlConn.Close()
            End If

        End Try

    End Function

        Dim SqlConn As New SqlConnection(strVcnetConn)
        Dim sqlstmt As String
        Dim result(0) As String
        Dim x As Integer
        Dim y As Integer

        Try
            result(0) = "none"
            sqlstmt = "SELECT cell, worknumber " & _
                        "FROM people " & _
                        "WHERE peopleid = " & pplid

            Dim sqlCmd As New SqlCommand(sqlstmt, SqlConn)
            sqlCmd.CommandType = CommandType.Text
            SqlConn.Open()
            sqlDr = sqlCmd.ExecuteReader

            While sqlDr.Read
                For x = 0 To 1
                    If LCase(sqlDr.GetValue(x).GetType.ToString) <> "system.dbnull" Then
                        ReDim Preserve result(y)
                        result(y) = sqlDr.GetString(x)
                        y += 1
                    End If
                Next
            End While

            sqlDr.Close()

            Return result

        Catch ex As SqlException
            Dim equip(2) As String
            equip(0) = "Error"
            equip(1) = CStr(ex.Number)
            equip(2) = ex.Message

            Return equip

        Finally
            If SqlConn.State = ConnectionState.Open Then
                SqlConn.Close()
            End If

        End Try

    End Function

    <WebMethod()> Public Function GetEquipReturn(ByVal callid As Integer) As String()

        Dim SqlConn As New SqlConnection(strDcdConn)
        Dim sqlstmt As String

        Try
            Dim equip(1) As String
            equip(0) = "None"
            sqlstmt = "SELECT dbo.parts.description + ' SN: ' + dbo.equipment.serialnumber AS equipitem, dbo.equipmentreturns.returnauthorization " & _
                        "FROM dbo.equipmenttopeople INNER JOIN " & _
                            "dbo.equipment ON dbo.equipmenttopeople.equipmentid = dbo.equipment.equipmentid INNER JOIN " & _
                            "dbo.parts ON dbo.equipment.partno = dbo.parts.partno INNER JOIN " & _
                            "dbo.equipmentreturns ON dbo.equipmenttopeople.equipmentid = dbo.equipmentreturns.equipmentid " & _
                        "WHERE dbo.equipmentreturns.callid = " & callid

            Dim sqlCmd As New SqlCommand(sqlstmt, SqlConn)
            sqlCmd.CommandType = CommandType.Text
            SqlConn.Open()
            sqlDr = sqlCmd.ExecuteReader

            While sqlDr.Read
                equip(0) = sqlDr.GetString(0)
                equip(1) = sqlDr.GetString(1)
            End While

            sqlDr.Close()

            Return equip

        Catch ex As SqlException
            Dim equip(2) As String
            equip(0) = "Error"
            equip(1) = CStr(ex.Number)
            equip(2) = ex.Message

            Return equip

        Finally
            If SqlConn.State = ConnectionState.Open Then
                SqlConn.Close()
            End If

        End Try

    End Function

    <WebMethod()> Public Function AddTimesheets(ByVal sname As String, ByVal tsdate As Date, ByVal projid As Integer, ByVal ph As String, ByVal actvy As String, ByVal desc As String, ByVal hours As Double) As String

        Dim SqlConn As New SqlConnection(strDcdConn)
        Dim cnt As Integer = 0
        Dim sqlstmt As String
        Dim billable As Byte
        Dim bhours As Double

        Try
            SqlConn.Open()

            'check if document has already been added
            sqlstmt = "SELECT count(*) FROM timesheetitems " & _
                "WHERE screenname = '" & sname & "' " & _
                "AND date = '" & tsdate & "' " & _
                "AND projid = " & projid & " " & _
                "AND phase = '" & ph & "' " & _
                "AND activity = '" & actvy & "' "

            Dim sqlCmd1 As New SqlCommand(sqlstmt, SqlConn)
            sqlCmd1.CommandType = CommandType.Text
            cnt = CInt(sqlCmd1.ExecuteScalar.ToString)

            'add document if not already in database
            If cnt = 0 Then
                'check if activity is billable
                sqlstmt = "SELECT billable FROM activities " & _
                    "WHERE activity = '" & actvy & "' "

                Dim sqlCmd2 As New SqlCommand(sqlstmt, SqlConn)
                sqlCmd2.CommandType = CommandType.Text
                billable = CByte(sqlCmd2.ExecuteScalar.ToString)

                If billable = 1 Then
                    bhours = hours
                Else
                    bhours = 0
                End If

                sqlstmt = "INSERT INTO timesheetitems (screenname, date, projid, phase, activity, detaildescription, hours, billable ) " & _
                            "VALUES (" & sname & ", '" & tsdate & "', " & projid & "', '" & ph & "', '" & actvy & "', '" & desc & "', " & hours & ", " & bhours & ")"
                Dim sqlCmd3 As New SqlCommand(sqlstmt, SqlConn)
                sqlCmd3.CommandType = CommandType.Text
                sqlCmd3.ExecuteNonQuery()
                Return "True"
            Else
                Return "Dup"
            End If


        Catch ex As SqlException
            Return ex.Number & "^" & ex.Message

        Finally
            If SqlConn.State = ConnectionState.Open Then
                SqlConn.Close()
            End If

        End Try

    End Function
