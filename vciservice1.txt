    Private Function Crypt(ByRef pass As String, ByRef Strg As String) As String

        Dim b As Integer
        Dim j As String
        Dim h As String
        Dim i As Integer
        Dim a As Integer
        Dim tmp1 As String
        Dim tmp2 As String

        a = 1

        For i = 1 To Len(Strg)
            b = Asc(Mid(pass, a, 1))
            a = a + 1
            If a > Len(pass) Then a = 1
            tmp1 = Chr(Asc(Mid(Strg, i, 1)) Xor b)
            tmp2 = tmp2 & tmp1
        Next

        Strg = tmp2
        h = ""

        For i = 1 To Len(Strg)
            j = Hex(Asc(Mid(Strg, i, 1)))
            If Len(j) = 1 Then j = "0" & j
            h = h & j
        Next

        Return h

    End Function

        End If

        Try
            SqlConn.Open()
            sqlstmt = "SELECT count(*) FROM iresults " & _

            Dim sqlCmd1 As New SqlCommand(sqlstmt, SqlConn)

            If cnt = 0 Then
                ReDim result(0)

            ElseIf cnt = 1 Then
                Dim dr As SqlClient.SqlDataReader
                ReDim result(2)

                sqlstmt = "SELECT CONVERT(varchar(10), dbo.iresults.ins_peopleid) + '^' + dbo.inspectors.initials, comment, inspectiondate " & _
                                "FROM dbo.iresults INNER JOIN " & _
                                    "dbo.inspectors ON dbo.iresults.ins_peopleid = dbo.inspectors.peopleid " & _
                                "WHERE projectid = " & projid & " " & _
                                "AND iresultsid = '" & item & "' "

                Dim sqlCmd2 As New SqlCommand(LCase(sqlstmt), SqlConn)
                sqlCmd2.CommandType = CommandType.Text

                dr = sqlCmd2.ExecuteReader

                While dr.Read
                    result(0) = dr.GetValue(0).ToString
                    result(1) = dr.GetValue(1).ToString
                    result(2) = dr.GetValue(2).ToString
                End While

                ReDim result(2)

            result(1) = CStr(ex.Number)
            result(2) = ex.Message

            Return result

        Finally
            If SqlConn.State = ConnectionState.Open Then
                SqlConn.Close()
            End If
